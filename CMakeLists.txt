cmake_minimum_required(VERSION 3.16)
project(ETCH VERSION 1.04.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(etch STATIC engine/etch.cpp engine/harmony.cpp engine/transfer.cpp engine/boot_state.cpp engine/watchdog.cpp target/r2_app.cpp target/q15_oscillator.cpp target/q15_voice.cpp target/q15_runtime.cpp firmware/fm1-r2-test/fm1_platform_port.cpp)
target_include_directories(etch PUBLIC engine hal target)
add_executable(etch_render engine/render.cpp)
target_link_libraries(etch_render PRIVATE etch)
add_executable(etch_test tests/test.cpp)
target_link_libraries(etch_test PRIVATE etch)
add_executable(etch_target_budget tools/target_budget.cpp)
target_link_libraries(etch_target_budget PRIVATE etch)
enable_testing()
add_test(NAME etch_core COMMAND etch_test)
add_test(NAME etch_target_budget COMMAND etch_target_budget)

add_executable(etch_runtime_test tests/test_runtime.cpp)
target_link_libraries(etch_runtime_test PRIVATE etch)
add_test(NAME etch_runtime COMMAND etch_runtime_test)
